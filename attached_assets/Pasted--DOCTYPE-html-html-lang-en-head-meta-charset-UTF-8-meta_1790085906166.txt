    <script>
        async function renderMap() {
            try {
                // Fetch pre-projected US map data (Albers USA projection mapped to 975x610)
                const us = await d3.json("https://cdn.jsdelivr.net/npm/us-atlas@3/states-10m.json");
                
                // Exclude Alaska (02), Hawaii (15), and territories to only show the contiguous US
                const nonContiguousIds = new Set(["02", "15", "60", "66", "69", "72", "78"]);
                const contiguousStates = topojson.feature(us, us.objects.states).features.filter(d => !nonContiguousIds.has(d.id));

                const svg = d3.select("#map-svg");
                const path = d3.geoPath(); // Uses identical pixel coordinates as the pre-projected JSON

                // Draw states with base fill and visible state lines
                svg.append("g")
                    .selectAll("path")
                    .data(contiguousStates)
                    .join("path")
                    .attr("d", path)
                    .attr("fill", "#1c1f26") // Dark slate background for land
                    .attr("stroke", "rgba(255, 255, 255, 0.15)") // Clear but subtle state borders
                    .attr("stroke-width", 1)
                    .attr("stroke-linejoin", "round");

                // Offscreen canvas logic ensures dots only spawn on valid land masses
                const offscreen = document.createElement("canvas");
                offscreen.width = 975;
                offscreen.height = 610;
                const offCtx = offscreen.getContext("2d", { willReadFrequently: true });
                const offPath = d3.geoPath().context(offCtx);

                // Draw ONLY the contiguous states onto the hidden canvas in solid white
                offCtx.fillStyle = "#ffffff";
                offCtx.beginPath();
                contiguousStates.forEach(feature => offPath(feature));
                offCtx.fill();

                // Extract pixel data to test x/y coordinates against state boundaries
                const imgData = offCtx.getImageData(0, 0, 975, 610).data;

                const canvas = document.getElementById("map-canvas");
                const ctx = canvas.getContext("2d");
                
                const targetDots = 60000;
                let dotsPlaced = 0;
                const dotRadius = 0.7; // Tiny radius to allow for heavy density without clumping entirely
                
                // Color configuration for dots: vibrant red with slight transparency for overlapping density effects
                ctx.fillStyle = "rgba(255, 40, 40, 0.5)"; 
                
                // Use a single path to render all dots simultaneously for maximum performance
                ctx.beginPath();

                // Rejection sampling: Pick a random coordinate, verify it's over a state, then keep it.
                while (dotsPlaced < targetDots) {
                    const x = Math.floor(Math.random() * 975);
                    const y = Math.floor(Math.random() * 610);
                    
                    // Check alpha channel of the offscreen canvas (index format: r, g, b, a)
                    const alphaIndex = (y * 975 + x) * 4 + 3;
                    
                    if (imgData[alphaIndex] > 0) {
                        // Point is inside a state boundary
                        ctx.moveTo(x, y);
                        ctx.arc(x, y, dotRadius, 0, Math.PI * 2);
                        dotsPlaced++;
                    }
                }
                
                ctx.fill();

                // Fade out loading screen once generation is complete
                const loader = document.getElementById('loading');
                loader.style.opacity = '0';
                setTimeout(() => loader.style.display = 'none', 500);

            } catch (error) {
                console.error("Error loading map data:", error);
                document.getElementById('loading').innerHTML = "<span class='text-red-500'>Failed to load map data.</span>";
            }
        }

        // Initialize rendering on load
        window.addEventListener("DOMContentLoaded", renderMap);
    </script>
